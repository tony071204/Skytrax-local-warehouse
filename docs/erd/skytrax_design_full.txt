dim_airline
-----------
airline_key INTEGER PK
airline_id BIGINT # original id from stg__airlines
airline_name VARCHAR
service_model VARCHAR NULL # full-service / low-cost, from seed (NOT FINAL)
region VARCHAR NULL # from seed (NOT FINAL)
peer_group VARCHAR NULL # from seed (NOT FINAL)
